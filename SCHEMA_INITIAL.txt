type Account {
  id_account gen_uuid() identity

  email string unique
  password_hash string
  role enum("LEARNER", "EDITOR", "ADMIN")
}

type User {
  id_user uuid ref Account::id_account identity

  name string
  user_name string unique
  native_dialect Region
  total_xp int default 0
  current_level int default 1
  current_streak int default 0
  longest_streak int default 0
  last_activity_at datetime default now()
  multiple achievement_list Achievement {
    unlocked_at datetime default now()
  }

  created_at datetime default now()
}

type Region {
  id_region gen_uuid() identity

  name string
  parent_zone string
  country_code string
  flag_emoji string
}

type Word {
  id_word int auto_increment identity
  id_region? uuid ref Region::id_region
  id_editor uuid ref User::id_user

  name string
  short_definition string
  definition string
  friendly_blurb string
  part_of_speech string
  image_url string
  gender enum("FEMALE", "MALE", "NEUTRAL")
  cefr_level enum("A1", "A2", "B1", "B2", "C1", "C2", "U")
  state enum("DRAFT", "APPROVED", "HIDDEN") default "DRAFT"
  cultural_note string
  is_taboo_or_vulgar bool

  readonly created_at datetime default now()
  updated_at datetime
}

type WordApproval {
  id_word int ref Definition::id_word
  id_editor uuid ref User::id_user

  approved_at datetime default now()

  trigger on insert for each {
    if select count(WordApproval)
      filter :id == self:id_word > 2 {
      update Definition filter :id = self:id_word
      set { state = "APPROVED" }
    }
  }

  identity (id_word, id_editor)
}

type Sense {
  id_sense int auto_increment identity
  id_word ref Definition::id_word
  id_region ref Region::id_region
  id_editor uuid ref User::id_user

  usage_frequency enum(
    "EVERYDAY", "COMMON", "RARE", "ARCHAIC"
  )
  definition string

  readonly created_at datetime default now()
  updated_at datetime
}

type Example {
  id_example int auto_increment identity
  id_word ref Definition::id_word
  id_editor ref User::id_user

  example string
  source string

  readonly created_at datetime default now()
  updated_at datetime
}

type Deck {
  id_deck int auto_increment identity
  id_user ref User::id_user

  title string
  description string
  is_public bool
  is_official bool

  readonly created_at datetime default now()
  updated_at datetime
}

type Collection {
  id_collection int auto_increment identity
  id_user ref User::id_user

  title string
  description string
  multiple deck_list Deck
  is_public bool
  is_featured bool
  is_official bool

  readonly created_at datetime default now()
  updated_at datetime
}

type XPTransaction {
  id_transaction int auto_increment identity
  id_user uuid ref User::id_user

  amount_xp int
  reason enum(
    "QUIZ_CORRECT",
    "MATCH_WIN",
    "DAILY_STREAK",
    "ACHIEVEMENT_UNLOCKED"
  )

  created_at datetime default now()
}

type Achievement {
  id_acchievement int auto_increment identity

  slug string unique
  title string
  description string
  badge_icon_url string
  tier enum("BRONZE", "SILVER", "GOLD", "ONYX")
  category enum(
    "STREAK", "COMPETITION", "MASTERY", "DIALECT_EXPLORER"
  )

  created_at datetime now()
}

type Mastery {
  id_user uuid ref User::id_user
  id_word int ref Definition::id_word

  progress_count int default 0
  last_reviewed_at datetime default now()

  identity (id_user, id_word)
}

type Match {
  id_match int auto_increment identity
  id_deck int ref Deck::id_deck

  room_code string
  status enum("WAITING", "IN_PROGRESS", "FINISHED")
  winner_user_id? uuid ref User::id_user
  is_public bool
  total_rounds int default 10

  created_at datetime default now()
  finished_at datetime
}

type MatchParticipant {
  id_match uuid ref Match::id_match
  id_user uuid ref User::id_user

  score int default 0
  rank int
  correct_answers int default 0
  total_time_spent_ms int

  identity (id_match, id_user)
}
